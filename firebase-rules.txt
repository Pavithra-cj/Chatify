rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    //user profile collection
  	match /user/{userId} {
      allow create: if request.auth != null && request.auth.uid == userId;
      allow read: if request.auth != null;
      
      // Allow updates to the document if:
      // 1. The user is updating their own document OR
      // 2. The only change is adding/removing the requesting user's ID to/from the friends array
      allow update: if request.auth != null && (
        request.auth.uid == userId || 
        (
          // Only the friends field is being changed
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['friends']) &&
          // The difference in the friends array is only the requesting user's ID
          request.resource.data.friends.removeAll(resource.data.friends).hasOnly([request.auth.uid])
        )
      );
      
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    
    // chats collection
    match /chats/{userId} {
      match /{otherUserId}/{messageId} {
        allow create: if request.auth != null &&
          (request.auth.uid == userId || request.auth.uid == otherUserId);
        
        allow read: if request.auth != null &&
          (request.auth.uid == userId || request.auth.uid == otherUserId);
      }
    }
    
    // Recent chats collection
    match /recent_chats/{userId} {
      match /messages/{otherUserId} {
        allow create, update: if request.auth != null && 
        	(request.auth.uid == userId || request.auth.uid == otherUserId);
        
        allow read: if request.auth != null && 
        	(request.auth.uid == userId || request.auth.uid == otherUserId);
      }
    }
  }
}
